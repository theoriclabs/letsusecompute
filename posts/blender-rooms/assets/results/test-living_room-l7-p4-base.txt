def create_material(name, color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = color
    bsdf.inputs['Roughness'].default_value = 0.7
    bsdf.inputs['Metallic'].default_value = 0.0
    return mat

def create_mesh(name, vertices, edges, faces, material_name):
    obj = bpy.data.objects.new(name, None)
    bpy.context.collection.objects.link(obj)
    
    mesh = bpy.data.meshes.new(name)
    mesh.from_pydata(vertices, edges, faces)
    mesh.update_calc()
    
    # Apply scale to fix potential float issues
    obj.scale = (1, 1, 1)
    
    mat = bpy.data.materials.get(material_name)
    if not mat:
        mat = create_material(name + "_mat", (0.5, 0.5, 0.5))
    
    obj.data.materials.append(mat)
    return obj

def create_box(name, loc, size, mat_name):
    verts = [
        (0, 0, 0), (size[0], 0, 0), (size[0], size[1], 0), (0, size[1], 0),
        (0, 0, size[2]), (size[0], 0, size[2]), (size[0], size[1], size[2]), (0, size[1], size[2])
    ]
    edges = [
        (0, 1), (1, 2), (2, 3), (3, 0),
        (4, 5), (5, 6), (6, 7), (7, 4),
        (0, 4), (1, 5), (2, 6), (3, 7)
    ]
    faces = [(0, 1, 2, 3), (4, 5, 6, 7), (0, 1, 5, 4), (2, 3, 7, 6), (0, 3, 7, 4), (1, 2, 6, 5)]
    return create_mesh(name, verts, edges, faces, mat_name)

def create_cylinder(name, loc, radius, height, segments=16, mat_name=None):
    verts = []
    edges = []
    faces = []
    
    for i in range(segments + 1):
        angle = 2 * math.pi * i / segments
        x = radius * math.cos(angle)
        z = radius * math.sin(angle)
        verts.append((x, 0, z))
        verts.append((x, height, z))
    
    for i in range(segments):
        v1 = i
        v2 = i + 1
        v3 = v2 + segments
        v4 = v1 + segments
        edges.append((v1, v2))
        edges.append((v3, v4))
        edges.append((v1, v4))
        edges.append((v2, v3))
        faces.append((v1, v2, v3, v4))
        
    obj = create_mesh(name, verts, edges, faces, mat_name if mat_name else "cyl_mat")
    obj.location = loc
    return obj
