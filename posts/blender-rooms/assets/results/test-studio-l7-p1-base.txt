import bpy
import math

# Clear existing objects to ensure a clean start
bpy.ops.object.select_all(action='SELECT')
bpy.ops.object.delete(use_global=False)

# --- Configuration ---
COLORS = {
    'dusty_blue': (0.529, 0.620, 0.678),   # 859eae
    'cream': (0.933, 0.902, 0.824),        # ece3d0
    'warm_wood': (0.608, 0.451, 0.365),    # 9b7359
    'accent_terra': (0.796, 0.553, 0.467), # cb8d74
}

# Helper to create a mesh with bevels
def create_beveled_box(location, size, color, bevel_depth=0.01, bevel_width=0.005, name_prefix=""):
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    obj = bpy.context.active_object
    obj.name = f"{name_prefix}_box"
    obj.data.name = f"{name_prefix}_mesh"
    obj.data.color_attributes.new(name="Color", domain='POINT', namespace="mat")
    obj.data.color_attributes["Color"].data[0].color = color
    
    # Apply bevel
    bpy.ops.object.shade_smooth()
    bpy.ops.object.modifier_add(type='BEVEL')
    obj.modifiers["Bevel"].depth = bevel_depth
    obj.modifiers["Bevel"].width = bevel_width
    obj.modifiers["Bevel"].segments = 1
    
    return obj

# Helper to create a plane (for rugs, mats)
def create_plane(location, size, color, name_prefix=""):
    bpy.ops.mesh.primitive_plane_add(size=1, location=location)
    obj = bpy.context.active_object
    obj.name = f"{name_prefix}_plane"
    obj.data.name = f"{name_prefix}_mesh"
    obj.data.color_attributes.new(name="Color", domain='POINT', namespace="mat")
    obj.data.color_attributes["Color"].data[0].color = color
    bpy.ops.object.shade_smooth()
    return obj
