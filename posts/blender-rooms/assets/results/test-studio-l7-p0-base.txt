def create_mesh(name, verts, faces, parent=None):
    """Helper to create a mesh from vertex and face lists."""
    obj = bpy.data.objects.new(name, bpy.data.meshes.new(name))
    if parent:
        obj.parent = parent
    bpy.context.collection.objects.link(obj)
    
    mesh = obj.data
    mesh.from_pydata(verts, [], faces)
    mesh.update()
    
    # Apply scale to fix potential zero-area faces issues
    obj.scale = (1, 1, 1)
    bpy.ops.object.transform_apply(scale=True)
    
    return obj

def create_material(name, color):
    mat = bpy.data.materials.new(name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = color
    bsdf.inputs['Roughness'].default_value = 0.7
    bsdf.inputs['Metallic'].default_value = 0.0
    return mat

# --- Furniture Helpers ---
def create_box(name, loc, size, mat_color, bevel=0.02):
    verts = [
        (-size[0]/2, -size[1]/2, -size[2]/2), (size[0]/2, -size[1]/2, -size[2]/2),
        (size[0]/2, size[1]/2, -size[2]/2), (-size[0]/2, size[1]/2, -size[2]/2),
        (-size[0]/2, -size[1]/2, size[2]/2), (size[0]/2, -size[1]/2, size[2]/2),
        (size[0]/2, size[1]/2, size[2]/2), (-size[0]/2, size[1]/2, size[2]/2)
    ]
    faces = [
        (0, 1, 2), (0, 2, 3), # Bottom
        (4, 5, 1), (4, 1, 0), # Front
        (5, 6, 2), (5, 2, 1), # Right
        (6, 7, 3), (6, 3, 2), # Back
        (7, 4, 6), (7, 6, 5), # Left
    ]
    obj = create_mesh(name, verts, faces)
    obj.location = loc
    obj.data.materials.append(create_material(name, mat_color))
    return obj

def create_cylinder(name, loc, radius, height, segments=8, mat_color=None):
    verts = []
    faces = []
    for i in range(segments + 1):
        angle = 2 * math.pi * i / segments
        x = radius * math.cos(angle)
        z = radius * math.sin(angle)
        verts.append((x, 0, z))
        verts.append((x, height, z))
    
    for i in range(segments):
        a = i
        b = (i + 1) % (segments + 1)
        c = b + (segments + 1)
        d = a + (segments + 1)
        faces.append((a, b, d))
        faces.append((b, c, d))
    
    obj = create_mesh(name, verts, faces)
    obj.location = loc
    if mat_color:
        obj.data.materials.append(create_material(name, mat_color))
    return obj
