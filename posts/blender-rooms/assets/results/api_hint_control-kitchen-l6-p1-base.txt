import bpy
import math

# Clear existing objects to ensure a clean start
bpy.ops.object.select_all(action='SELECT')
bpy.ops.object.delete(use_global=False)

# Define Colors (RGBA)
C_DUSTY_BLUE = (0.525, 0.616, 0.647, 1.0)   # 859eae
C_CREAM = (0.941, 0.902, 0.824, 1.0)        # ece3d0
C_WOOD = (0.608, 0.451, 0.333, 1.0)         # 9b7359
C_ACCENT = (0.796, 0.553, 0.459, 1.0)       # cb8d74

def create_material(name, color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs["Base Color"].default_value = color
    bsdf.inputs["Roughness"].default_value = 0.7
    bsdf.inputs["Metallic"].default_value = 0.0
    return mat

def create_cube(location, size, name, mat):
    obj = bpy.data.objects.new(name, bpy.data.meshes.new(name))
    obj.location = location
    obj.dimensions = size
    bpy.context.collection.objects.link(obj)
    obj.data.materials.append(mat)
    return obj

def create_cylinder(location, radius, height, name, mat, segments=16):
    obj = bpy.data.objects.new(name, bpy.data.meshes.new(name))
    obj.location = location
    me = obj.data
    me.from_meshdata(bpy.data.meshes.new(name))
    me.from_object(bpy.data.meshes.new("temp_cyl"))
    # Re-create cylinder properly
    me = bpy.data.meshes.new(name)
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=height, location=location, segments=segments)
    bpy.context.collection.objects.link(obj)
    obj.data = me
    obj.data.materials.append(mat)
    return obj

def create_box(location, size, name, mat):
    obj = bpy.data.objects.new(name, bpy.data.meshes.new(name))
    obj.location = location
    obj.dimensions = size
    bpy.context.collection.objects.link(obj)
    obj.data.materials.append(mat)
    return obj

# --- Bevels ---
def add_bevel(obj, thickness=0.01, width=0.01):
    mod = obj.modifiers.new(name="edge", type="BEVEL")
    mod.width = width
    mod.limit_method = 'ANGLE'
    mod.angle_limit = math.radians(30)
    mod.bevel_depth = thickness

# Apply bevels to furniture parts
for obj in [ref_door_l, ref_door_r, counter_top, stove_top, sink_rim, tap_spout, table_top, bowl, plant_leaves]:
    add_bevel(obj)
