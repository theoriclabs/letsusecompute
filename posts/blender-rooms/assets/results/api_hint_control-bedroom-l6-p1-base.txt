# Helper function to create a beveled cube
def create_beveled_cube(location, size, color, name, bevel=0.02):
    obj = bpy.data.objects.new(name, bpy.data.meshes.new(name))
    obj.location = location
    bpy.context.collection.objects.link(obj)
    
    mesh = obj.data
    mat = bpy.data.materials.new(name=f"{name}_mat")
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs["Base Color"].default_value = (color[0], color[1], color[2], 1.0)
    mat.diffuse_color = (color[0], color[1], color[2], 1.0) # Compatibility note
    obj.data.materials.append(mat)
    
    # Create cube
    bpy.ops.mesh.primitive_cube_add(size=1, location=location)
    bpy.context.active_object.name = name
    bpy.context.view_layer.objects.active = obj
    
    # Apply bevel
    mod = obj.modifiers.new(name="bevel", type="BEVEL")
    mod.width = bevel
    mod.segments = 1
    
    return obj

# Helper function to create a beveled plane (for rug, art)
def create_beveled_plane(location, size, color, name, bevel=0.02):
    obj = bpy.data.objects.new(name, bpy.data.meshes.new(name))
    obj.location = location
    bpy.context.collection.objects.link(obj)
    
    mesh = obj.data
    mat = bpy.data.materials.new(name=f"{name}_mat")
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs["Base Color"].default_value = (color[0], color[1], color[2], 1.0)
    mat.diffuse_color = (color[0], color[1], color[2], 1.0)
    obj.data.materials.append(mat)
    
    bpy.ops.mesh.primitive_plane_add(size=1, location=location)
    bpy.ops.object.transform_scale(scale=(size[0], size[1], 1), center='MEDIAN')
    bpy.context.active_object.name = name
    bpy.context.view_layer.objects.active = obj
    
    mod = obj.modifiers.new(name="bevel", type="BEVEL")
    mod.width = bevel
    mod.segments = 1
    
    return obj
