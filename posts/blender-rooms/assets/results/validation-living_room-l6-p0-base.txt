# --- Helper Functions ---

def create_cube(name, loc, size, color, bevel=0.01):
    """Creates a low-poly cube with a small bevel."""
    obj = bpy.data.objects.new(name, bpy.data.meshes.new(name))
    obj.location = loc
    obj.rotation_euler = (math.radians(90), 0, 0)  # Align Z up
    obj.active_material = create_material(color)
    
    mesh = obj.data
    verts = [
        (size[0]/2, size[1]/2, size[2]/2),
        (-size[0]/2, size[1]/2, size[2]/2),
        (-size[0]/2, -size[1]/2, size[2]/2),
        (size[0]/2, -size[1]/2, size[2]/2),
        (size[0]/2, size[1]/2, -size[2]/2),
        (-size[0]/2, size[1]/2, -size[2]/2),
        (-size[0]/2, -size[1]/2, -size[2]/2),
        (size[0]/2, -size[1]/2, -size[2]/2)
    ]
    faces = [
        (0, 1, 2, 3), # Top
        (5, 4, 0, 1), # Front
        (5, 1, 2, 6), # Right
        (3, 2, 6, 7), # Back
        (7, 6, 5, 4), # Left
        (3, 0, 4, 7)  # Bottom
    ]
    
    mesh.from_pydata(verts, [], faces)
    mesh.update()
    
    # Apply bevel modifier
    mod = mesh.modifiers.new(name="Bevel", type='BEVEL')
    mod.width = bevel
    mod.angle = math.radians(45)
    mod.limit_method = 'ANGLE'
    mod.limit_angle = math.radians(45)
    
    bpy.context.scene.objects.link(obj)
    return obj

def create_material(color_hex):
    mat = bpy.data.materials.new(name=f"mat_{color_hex}")
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = [int(x, 16)/255 for x in color_hex[1:].hex()]
    bsdf.inputs['Roughness'].default_value = 0.7
    bsdf.inputs['Metallic'].default_value = 0.0
    return mat
