# Define Colors (RGBA)
C_SAGE = (0.5686, 0.6745, 0.5373, 1.0)   # 91aa8a
C_OAK = (0.6078, 0.4392, 0.3020, 1.0)    # 9b704d
C_CREAM = (0.8863, 0.7255, 0.4196, 1.0)  # e2b96b
C_WARM = (0.9176, 0.8471, 0.8000, 1.0)   # e8dfcd (approximated warm beige)

def create_material(name, color):
    mat = bpy.data.materials.new(name=name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs["Base Color"].default_value = color
    bsdf.inputs["Roughness"].default_value = 0.6
    bsdf.inputs["Metallic"].default_value = 0.0
    return mat

def create_mesh(name, verts, edges, faces, loc, rot, scale, mat_name):
    obj = bpy.data.objects.new(name, None)
    bpy.context.collection.objects.link(obj)
    mesh = bpy.data.meshes.new(name)
    mesh.from_pydata(verts, edges, faces)
    mesh.update()
    obj.data = mesh
    obj.location = loc
    obj.rotation_euler = rot
    obj.scale = scale
    if mat_name:
        mat = bpy.data.materials.get(mat_name)
        if not mat:
            mat = create_material(mat_name, C_OAK)
        obj.data.materials.append(mat)
    return obj

def add_bevel(obj, width=0.01):
    mod = obj.modifiers.new(name="bevel", type="BEVEL")
    mod.width = width
    mod.angle = math.radians(30)
    mod.limit_method = 'ANGLE'
    mod.limit_angle = math.radians(30)

# --- Helper: Box with bevel ---
def make_box(name, size, loc, rot, scale, mat, bevel=True):
    verts = []
    edges = []
    faces = []
    
    x, y, z = size
    # Vertices
    v0 = [ -x/2, -y/2, -z/2 ]
    v1 = [  x/2, -y/2, -z/2 ]
    v2 = [  x/2,  y/2, -z/2 ]
    v3 = [ -x/2,  y/2, -z/2 ]
    v4 = [ -x/2, -y/2,  z/2 ]
    v5 = [  x/2, -y/2,  z/2 ]
    v6 = [  x/2,  y/2,  z/2 ]
    v7 = [ -x/2,  y/2,  z/2 ]
    verts.extend([v0, v1, v2, v3, v4, v5, v6, v7])
    
    # Edges
    edges.extend([ (0,1), (1,2), (2,3), (3,0),
                   (4,5), (5,6), (6,7), (7,4),
                   (0,4), (1,5), (2,6), (3,7) ])
    
    # Faces
    faces.extend([ (0,1,2,3), (4,5,6,7), (0,1,5,4), (1,2,6,5), (2,3,7,6), (3,0,7,4) ])
    
    obj = create_mesh(name, verts, edges, faces, loc, rot, scale, mat)
    if bevel:
        add_bevel(obj)
    return obj
